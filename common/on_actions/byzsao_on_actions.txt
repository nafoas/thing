# Redraws the 1936 Middle East at game start.
# Ownership is changed by annexing whole tags and matching state cores,
# so no vanilla state files are overridden.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = byzsao_world_setup } }
				set_global_flag = byzsao_world_setup

				# ---- The Roman Proletarian Republic ----
				TUR = {
					if = { limit = { country_exists = GRE } annex_country = { target = GRE transfer_troops = no } }
					if = { limit = { country_exists = EGY } annex_country = { target = EGY transfer_troops = no } }
					if = { limit = { country_exists = SYR } annex_country = { target = SYR transfer_troops = no } }
					if = { limit = { country_exists = LEB } annex_country = { target = LEB transfer_troops = no } }
					if = { limit = { country_exists = PAL } annex_country = { target = PAL transfer_troops = no } }
					if = { limit = { country_exists = JOR } annex_country = { target = JOR transfer_troops = no } }
				}
				# Levant and Egypt held by Britain and France in vanilla
				every_state = {
					limit = {
						NOT = { is_owned_by = TUR }
						OR = {
							is_core_of = EGY
							is_core_of = SYR
							is_core_of = LEB
							is_core_of = PAL
							is_core_of = JOR
							is_core_of = GRE
						}
					}
					TUR = { transfer_state = PREV }
				}
				TUR = {
					every_owned_state = {
						limit = { NOT = { is_core_of = TUR } }
						add_core_of = TUR
					}
				}

				# ---- The Saoshyantate of Eranshahr ----
				PER = {
					if = { limit = { country_exists = IRQ } annex_country = { target = IRQ transfer_troops = no } }
					if = { limit = { country_exists = AFG } annex_country = { target = AFG transfer_troops = no } }
					if = { limit = { country_exists = SAU } annex_country = { target = SAU transfer_troops = no } }
					if = { limit = { country_exists = YEM } annex_country = { target = YEM transfer_troops = no } }
					if = { limit = { country_exists = OMA } annex_country = { target = OMA transfer_troops = no } }
				}
				every_state = {
					limit = {
						NOT = { is_owned_by = PER }
						OR = {
							is_core_of = IRQ
							is_core_of = SAU
							is_core_of = YEM
							is_core_of = OMA
						}
					}
					PER = { transfer_state = PREV }
				}
				PER = {
					every_owned_state = {
						limit = { NOT = { is_core_of = PER } }
						add_core_of = PER
					}
				}

				# Tell the players what world they are in
				every_country = {
					limit = { is_ai = no }
					country_event = { id = byzsao.1 days = 1 }
				}
			}
		}
	}
}
